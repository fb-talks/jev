  model: "google/gemini-3.8-flash",
  instructions: "Write 3 replies: curt, friendly, detailed",
  prompt: comment,
  output: Output.object({ schema: repliesSchema }),
});
// always three replies as data: no text to parse
2 · JEV SCORES THEM, IN PARALLEL
const questions = {
  quality: score("How helpful and polite is this reply?", levels),
};
// 3 calls at once: the time is the slowest one, not the sum
const results = await Promise.all(replies.map((reply) =>
  jev.systemOne({ state: `${comment}\n${reply}`, questions })));
3 · OUR CODE PICKS THE BEST
const scores = results.map((r) => r.answers.quality.score / 3);
